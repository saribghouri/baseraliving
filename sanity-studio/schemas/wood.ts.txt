import { defineField, defineType } from 'sanity';

export default defineType({
  name: 'wood',
  title: 'Timber',
  type: 'document',
  fields: [
    defineField({ name: 'name', title: 'Name', type: 'string', validation: (r) => r.required() }),
    defineField({ name: 'localName', title: 'Local name', type: 'string', description: 'e.g. Akhrot, Tali' }),
    defineField({ name: 'slug', title: 'URL slug', type: 'slug', options: { source: 'name' }, validation: (r) => r.required() }),
    defineField({ name: 'image', title: 'Grain photograph', type: 'image', options: { hotspot: true } }),
    defineField({ name: 'origin', title: 'Origin', type: 'string' }),
    defineField({ name: 'hardness', title: 'Hardness', type: 'string', description: 'e.g. Janka 1010 lbf — medium hard' }),
    defineField({ name: 'tone', title: 'Tone', type: 'string' }),
    defineField({ name: 'grain', title: 'Grain', type: 'string' }),
    defineField({ name: 'bestFor', title: 'Best for', type: 'array', of: [{ type: 'string' }], options: { layout: 'tags' } }),
    defineField({ name: 'notes', title: 'Notes', type: 'text', rows: 4 }),
    defineField({
      name: 'priceIndex', title: 'Relative cost', type: 'number',
      options: { list: [1, 2, 3, 4] }, initialValue: 2,
    }),
    defineField({ name: 'swatchFrom', title: 'Swatch colour (dark)', type: 'string', initialValue: '#4A3324' }),
    defineField({ name: 'swatchTo', title: 'Swatch colour (light)', type: 'string', initialValue: '#7A5C3E' }),
  ],
  preview: { select: { title: 'name', subtitle: 'localName', media: 'image' } },
});
